def match(doc: str, quote: str):
    """Return the name of the loosest fold that matches, or None."""
    for name, fn in (("exact", f_exact), ("whitespace", f_ws), ("case", f_case),
                     ("no-whitespace", f_nows), ("markdown", f_md), ("nfkc", f_nfkc),
                     ("typographic", f_typo), ("c-escape", f_cescape),
                     ("de-wrapped", f_dewrap), ("html", f_html),
                     ("md-link", f_mdlink)):
        try:
            if fn(doc, quote):
                return name
        except Exception:
            continue
    return None

def companion_urls(url: str, row):
    """Other URLs the same row legitimately points at.

    The URL is normalised first: a trailing quote in prose is not part of it.

    A row may cite several files of one repository (`audio_stream_mp3.h` and
    `.cpp`), but the gate block names one URL. A quote that lives in the
    sibling file is not a fabrication — it is a row that cites two files and
    a gate block that names one of them. The same holds for a J-STAGE
    article page whose quote is in the PDF, and for a repository row whose
    own-work artifact sits in the lane tree.
    """
    url = URL_TAIL_RE.sub("", url)
    out = []
    src = row["source"]
    # ONLY the sibling files the row itself names. Generating every extension
    # of every stem meant six URLs per row across 1,273 rows — most of them
    # unreachable, each costing a timeout, and the census never finished.
    named = set()
    for m in re.finditer(r"[\w./-]+\.(?:h|hpp|hh|c|cc|cpp|cxx|py|sh|rs|go|md|txt)", src):
        named.add(m.group(0).rsplit("/", 1)[-1])
    primary = url.rsplit("/", 1)[-1]
    for sib in sorted(named):
        if sib == primary:
            continue
        try:
            d = url.rsplit("/", 1)[0]
            cand = f"{d}/{sib}"
        except Exception:
            continue
        if cand not in out:
            out.append(cand)
    # J-STAGE: the PDF variant of an article page
    if "_article/" in url:
        out.append(url.replace("_article/", "_pdf/"))
    # A repository-root URL for an own-work row: the real evidence is local.
    if row.get("kind") == "own-work" and re.match(r"^https?://[^/]+/[^/]+/?$", url):
        out.append(url)
    return out
